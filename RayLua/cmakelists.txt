cmake_minimum_required(VERSION 3.21)
cmake_policy(VERSION 3.5)
project(BuildEditor2 C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})

option(DISABLE_LUA "Disable Lua support" ON)

# ── dependencies ──────────────────────────────────────────────────────────────
include(FetchContent)

# raylib 6.0
FetchContent_Declare(
    raylib
    URL https://github.com/raysan5/raylib/archive/refs/tags/6.0.tar.gz
    URL_HASH SHA256=2b3ee1e2120c7a0796b33062c7e9a694dd8a8caa56a96319ac8c8ecf54a90d0b
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES    OFF CACHE BOOL "" FORCE)
set(INSTALL_RAYLIB OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(raylib)

# imgui v1.92.7
FetchContent_Declare(
    imgui
    URL https://github.com/ocornut/imgui/archive/refs/tags/v1.92.7.tar.gz
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
FetchContent_MakeAvailable(imgui)

add_library(imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_demo.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp
)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)

# rlImGui — Raylib_6_0
FetchContent_Declare(
    rlimgui
    URL https://github.com/raylib-extras/rlImGui/archive/refs/tags/Raylib_6_0.tar.gz
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
FetchContent_MakeAvailable(rlimgui)

add_library(rlimgui STATIC ${rlimgui_SOURCE_DIR}/rlImGui.cpp)
target_link_libraries(rlimgui PUBLIC raylib imgui)
target_include_directories(rlimgui PUBLIC ${rlimgui_SOURCE_DIR})

# ── project sources ───────────────────────────────────────────────────────────
include(luajit.cmake)
include(shared.cmake)
include(raygame.cmake)
include(duke_target.cmake)

# ── resource copies ───────────────────────────────────────────────────────────
# Use a stamp file so the copy step is skipped when nothing has changed.
# add_custom_target is always considered out-of-date; add_custom_command with
# OUTPUT lets make track whether the stamp is older than any source file.
file(GLOB TILE_FILES "${CMAKE_SOURCE_DIR}/TILES???.ART")
file(GLOB SHADER_FILES "${CMAKE_SOURCE_DIR}/Shaders/*")
file(GLOB FONT_FILES   "${CMAKE_SOURCE_DIR}/fonts/*")
file(GLOB CONTENT_FILES "${CMAKE_SOURCE_DIR}/Content/*" "${CMAKE_SOURCE_DIR}/Content/**/*")

set(COPY_STAMP "${CMAKE_BINARY_DIR}/copy_content.stamp")

add_custom_command(
    OUTPUT  "${COPY_STAMP}"
    DEPENDS ${TILE_FILES} ${SHADER_FILES} ${FONT_FILES} ${CONTENT_FILES}
            "${CMAKE_SOURCE_DIR}/e3l3.map"
            "${CMAKE_SOURCE_DIR}/palette.dat"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/Shaders" "${CMAKE_BINARY_DIR}/Shaders"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/e3l3.map" "${CMAKE_BINARY_DIR}/e3l3.map"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/palette.dat" "${CMAKE_BINARY_DIR}/palette.dat"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/fonts" "${CMAKE_BINARY_DIR}/fonts"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/Content" "${CMAKE_BINARY_DIR}/Content"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${TILE_FILES} "${CMAKE_BINARY_DIR}"
    COMMAND ${CMAKE_COMMAND} -E touch "${COPY_STAMP}"
    COMMENT "Copying runtime content to build dir"
    VERBATIM
)
add_custom_target(copy_content ALL DEPENDS "${COPY_STAMP}")
add_dependencies(RayGame copy_content)
